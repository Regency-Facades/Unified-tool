function initSignaturePad() {
    const c = document.getElementById('sigCanvas');
    const getPos = (e) => { const r = c.getBoundingClientRect(); return { x: e.clientX - r.left, y: e.clientY - r.top }; };
    c.addEventListener('pointerdown', (e) => { e.preventDefault(); c.setPointerCapture(e.pointerId); sigDrawing = true; const p = getPos(e); const ctx = c.getContext('2d'); ctx.beginPath(); ctx.moveTo(p.x, p.y); });
    c.addEventListener('pointermove', (e) => { if (!sigDrawing) return; e.preventDefault(); const p = getPos(e); const ctx = c.getContext('2d'); ctx.lineTo(p.x, p.y); ctx.stroke(); });
    ['pointerup', 'pointercancel'].forEach(ev => c.addEventListener(ev, () => { sigDrawing = false; }));
}

}

function duplicateTopic(baseTopicStr, currentTopicStr, event) {
    event.stopPropagation(); event.preventDefault();
    const ref = prompt("Inspect another item of this type? \n\nEnter location / bay reference (e.g. 'Bay 19', 'Gridline C'):");
    if (!ref) return;

    const templateQs = DEFAULT_QUESTIONS_TEMPLATE.filter(q => q.baseTopic === baseTopicStr);
    const newTopicName = `${baseTopicStr.replace(' (TW Item)', '')} - Ref: [${ref}]`;

    const newQs = templateQs.map((q, i) => ({
        id: 'q_dup_' + Date.now() + '_' + i,
        baseTopic: q.baseTopic,
        topic: newTopicName,
        type: q.type,
        text: q.text,
        answer: "",
        severity: "48h",
        comment: "",
        photo: "",
        idx: 0
    }));

    let insertIndex = questionsList.length;
    for (let i = questionsList.length - 1; i >= 0; i--) {
        if (questionsList[i].topic === currentTopicStr) {
            insertIndex = i + 1;
            break;
        }
    }

    questionsList.splice(insertIndex, 0, ...newQs);
    reindexQuestions();
    renderQuestions();
    updateSequentialLocks();
    lucide.createIcons();
    alert(`Added: ${newTopicName}`);
}

function markTopicNA(topicStr, event) {
    event.stopPropagation(); event.preventDefault();
    if(!confirm(`Mark "${topicStr}" as Not Applicable?\n\nThis will bypass photo requirements for this category since the item is not present on site.`)) return;

    questionsList.forEach(q => {
        if (q.topic === topicStr) {
            q.answer = 'N/A';
            q.photo = 'SKIP'; 
            q.comment = 'Category marked N/A: Not present on site.';
        }
    });
    
    const detailsElements = document.querySelectorAll('details');
    detailsElements.forEach(d => {
        if(d.innerHTML.includes(topicStr)) d.removeAttribute('open');
    });

    renderQuestions();
    updateSequentialLocks();
    lucide.createIcons();
}

function updateSequentialLocks() {
    maxUnlockedIndex = 0;
    for (let i = 0; i < questionsList.length; i++) {
        if (questionsList[i].answer && (questionsList[i].photo || questionsList[i].photo === 'SKIP')) {
            maxUnlockedIndex = i + 1;
        } else {
            break;
        }
    }

    for (let i = 0; i < questionsList.length; i++) {
        const wrapper = document.getElementById(`q_wrapper_${i}`);
        if (!wrapper) continue;
        if (i <= maxUnlockedIndex) {
            wrapper.classList.remove('locked-item');
        } else {
            wrapper.classList.add('locked-item');
        }
    }

    if (maxUnlockedIndex < questionsList.length && maxUnlockedIndex > 0) {
        const currentTopicStr = questionsList[maxUnlockedIndex].topic;
        const prevTopicStr = questionsList[maxUnlockedIndex - 1].topic;
        if (currentTopicStr !== prevTopicStr) {
            const activeTopics = [...new Set(questionsList.map(q => q.topic))];
            const topicIdx = activeTopics.indexOf(currentTopicStr);
            if (topicIdx > 0) {
                const detailsEl = document.getElementById(`details_topic_${topicIdx}`);
                if (detailsEl && !detailsEl.open) detailsEl.open = true;
            }
        }
    }
}

function attemptSetAnswer(idx, val) {
    if (!questionsList[idx].photo && questionsList[idx].photo !== 'SKIP') {
        alert("Rule Enforcement: A photo is mandatory for EVERY TW question before logging an answer. Please capture or upload a photo first.");
        return;
    }

    questionsList[idx].answer = val;
    renderQuestions(); 
    updateSequentialLocks();
    lucide.createIcons();

    // Automatically scroll to the next question smoothly
    const nextIdx = idx + 1;
    const nextWrapper = document.getElementById(`q_wrapper_${nextIdx}`);
    if (nextWrapper) {
        nextWrapper.scrollIntoView({ behavior: 'smooth', block: 'center' });
    } else {
        const submitBtn = document.getElementById('submitBtn');
        if (submitBtn) submitBtn.scrollIntoView({ behavior: 'smooth', block: 'center' });
    }
}

function compressAndStorePhoto(idx, input) {
    if (input.files && input.files[0]) {
        const reader = new FileReader();
        reader.onload = function(e) {
            const img = new Image();
            img.onload = function() {
                const canvas = document.createElement('canvas');
                let width = img.width, height = img.height;
                // Higher resolution constraint for crystal clear PDF evidence
                if (width > 800) { height = Math.round((height * 800) / width); width = 800; }
                canvas.width = width; canvas.height = height;
                canvas.getContext('2d').drawImage(img, 0, 0, width, height);
                const compressedDataUrl = canvas.toDataURL('image/jpeg', 0.85);
                
                questionsList[idx].photo = compressedDataUrl;
                document.getElementById(`photo_thumb_img_${idx}`).src = compressedDataUrl;
                document.getElementById(`photo_thumb_box_${idx}`).classList.remove('hidden');

                updateSequentialLocks();
            };
            img.src = e.target.result;
        };
        reader.readAsDataURL(input.files[0]);
    }
}

function addImageToPdf(doc, imageStr, x, y, width, height) {
    if (!imageStr || typeof imageStr !== 'string' || imageStr.indexOf('data:image') !== 0) return false;
    if (imageStr === 'SKIP') return false;
    try {
        let fmt = 'JPEG';
        if (imageStr.indexOf('image/png') !== -1) fmt = 'PNG';
        else if (imageStr.indexOf('image/webp') !== -1) fmt = 'WEBP';
        doc.addImage(imageStr, fmt, x, y, width, height);
        return true;
    } catch (err) { return false; }
}

function generatePdfBlob(audit, c, s) {
    const { jsPDF } = window.jspdf;
    const doc = new jsPDF();
    const themeColor = branding.color || '#78350f';

    const tableRows = [];
    (audit.questions || []).forEach(q => {
        if (q.photo === 'SKIP') return;
        let result = q.answer === 'Yes' ? 'Compliant' : (q.answer === 'No' ? 'Defect' : 'N/A');
        let urgency = q.answer === 'No' ? (q.severity === 'Red' ? 'PROHIBITION' : 'MINOR') : '-';
        let findings = q.comment || 'No comment';
        
        tableRows.push([
            q.topic.replace(' (TW Item)', ''), 
            q.text, 
            result, 
            urgency, 
            findings, 
            q.photo ? { content: '', image: q.photo } : 'No Photo'
        ]);
    });

    doc.autoTable({
        startY: 65,
        head: [['Temporary Works Category', 'Criteria Checked', 'Result', 'Urgency', 'TWS Observations', 'Evidence Photo']],
        body: tableRows,
        headStyles: { fillColor: themeColor, fontSize: 8, fontStyle: 'bold' },
        styles: { fontSize: 7, cellPadding: 2.5, valign: 'middle', overflow: 'linebreak' },
        columnStyles: { 
            0: { cellWidth: 30 }, 
            1: { cellWidth: 50 }, 
            2: { cellWidth: 18 }, 
            3: { cellWidth: 20 }, 
            4: { cellWidth: 42 }, 
            5: { cellWidth: 26, minCellHeight: 22 } 
        },
        didDrawCell: function(data) {
            // Render crisp inline photo in table cell
            if (data.section === 'body' && data.column.index === 5 && data.cell.raw && typeof data.cell.raw === 'object' && data.cell.raw.image) {
                try {
                    let dim = 18;
                    doc.addImage(data.cell.raw.image, 'JPEG', data.cell.x + 4, data.cell.y + 2, dim, dim);
                } catch(e) {}
            }
        },
        didParseCell: function(data) {
            if (data.section === 'body' && data.column.index === 2) {
                if (data.cell.raw === 'Defect') { data.cell.styles.textColor = [220, 38, 38]; data.cell.styles.fontStyle = 'bold'; }
                if (data.cell.raw === 'Compliant') { data.cell.styles.textColor = [16, 185, 129]; }
            }
        },
        // Adds professional corporate header banner to every page automatically
        didDrawPage: function(data) {
            doc.setFillColor(themeColor);
            doc.rect(0, 0, 210, 26, 'F');
            doc.setTextColor(255, 255, 255);
            doc.setFontSize(14);
            doc.setFont("helvetica", "bold");
            doc.text(branding.title || 'Temporary Works Inspection UK', 14, 12);
            doc.setFontSize(8.5);
            doc.setFont("helvetica", "normal");
            doc.text(`Official Report | Project: ${c.name || 'N/A'} | Site: ${s.name || 'N/A'} | Date: ${audit.date}`, 14, 20);

            // Footer on every page
            doc.setFontSize(7);
            doc.setTextColor(150, 150, 150);
            doc.text("Official Temporary Works Inspection Certificate.", 14, 288);
            doc.text(`Page ${doc.internal.getNumberOfPages()}`, 185, 288);
        }
    });

    // APPENDIX: Supervisor Summary & Signatures
    doc.addPage();
    let yPos = 35;
    doc.setTextColor(40, 40, 40); 
    doc.setFontSize(11); 
    doc.setFont("helvetica", "bold");
    doc.text("TWS Executive Summary & Declarations", 14, yPos);
    yPos += 8;

    doc.setFontSize(9); doc.setFont("helvetica", "normal");
    const notesSplit = doc.splitTextToSize((audit.inspectorNotes || "No summary provided."), 180);
    doc.text(notesSplit, 14, yPos);
    yPos += (notesSplit.length * 5) + 10;

    doc.setFontSize(10); doc.setFont("helvetica", "bold");
    doc.text("TWS Sign-Off (Field):", 14, yPos);
    doc.text("Site Rep Acknowledgement:", 110, yPos);
    yPos += 5;

    if (audit.auditorSignature) addImageToPdf(doc, audit.auditorSignature, 14, yPos, 55, 22);
    if (audit.siteRepSignature) addImageToPdf(doc, audit.siteRepSignature, 110, yPos, 55, 22);
    yPos += 26;

    doc.setDrawColor(60, 60, 60); 
    doc.line(14, yPos, 78, yPos);
    doc.line(110, yPos, 174, yPos);
    yPos += 5;

    doc.setFontSize(8.5); doc.setFont("helvetica", "normal");
    doc.text('Name: ' + (audit.inspectorName || 'N/A') + ' (' + (audit.inspectorQuals || 'N/A') + ')', 14, yPos);
    doc.text('Name: ' + (audit.siteRepName || 'N/A'), 110, yPos);

    return doc.output('blob');
}

async function uploadPdfToSupabase(audit, pdfBlob) {
    if (!supabaseClient) return null;
    try {
        const fileName = `TW_Report_${audit.id}_${audit.date}.pdf`;
        const filePath = `reports/${fileName}`;

        const { data: uploadData, error: uploadError } = await supabaseClient.storage
            .from('tw_reports')
            .upload(filePath, pdfBlob, { contentType: 'application/pdf', upsert: true });

        if (uploadError) {
            console.warn("PDF Storage Notice: " + uploadError.message);
            return null;
        }

        const { data: urlData } = supabaseClient.storage.from('tw_reports').getPublicUrl(filePath);
        return urlData ? urlData.publicUrl : null;
    } catch(err) {
        console.warn("Storage Upload Error:", err);
        return null;
    }
}

        }
        if (!auditorSignature) {
            alert("⚠️ TWS Signature required: please sign the document before submitting.");
            return;
        }

        const unanswered = questionsList.findIndex(q => !q.answer || (!q.photo && q.photo !== 'SKIP'));
        if (unanswered !== -1) {
            alert(`⚠️ Incomplete Inspection: You stopped at question ${unanswered + 1}. You must complete all questions in sequence and provide a photo for each before submitting.`);
            return;
        }

        let totalValid = 0; let yesCount = 0; let hasCritical = false;
        questionsList.forEach(q => {
            if (q.answer !== 'N/A') {
                totalValid++;
                if (q.answer === 'Yes') yesCount++;
                if (q.answer === 'No' && q.severity === 'Red') hasCritical = true;
            }
        });
        let score = totalValid > 0 ? Math.round((yesCount / totalValid) * 100) : 0;
        let finalStatus = (score === 100 && !hasCritical) ? "Closed" : "Open";

        const clientObj = clients.find(c => c.id === clientId) || { name: 'Project' };
        const siteObj = sites.find(s => s.id === siteId) || { name: 'Site' };

        const newAudit = {
            id: 'a_' + Date.now(),
            clientId,
            siteId,
            date: new Date().toISOString().split('T')[0],
            inspectorName,
            inspectorQuals,
            inspectorNotes,
            auditorSignature,
            siteRepName: document.getElementById('inputSiteRepName').value.trim(),
            siteRepSignature: repSignature,
            score,
            status: finalStatus,
            questions: questionsList,
            pdfUrl: ""
        };

        // Generate PDF Blob
        const pdfBlob = generatePdfBlob(newAudit, clientObj, siteObj);

        // Upload PDF to Supabase Storage if configured
        if (supabaseClient) {
            const pdfUrl = await uploadPdfToSupabase(newAudit, pdfBlob);
            if (pdfUrl) newAudit.pdfUrl = pdfUrl;
        }

        let existing = JSON.parse(localStorage.getItem('tw_audits')) || [];
        existing.push(newAudit);
        try {
            localStorage.setItem('tw_audits', JSON.stringify(existing));
        } catch(e) { console.warn("Local storage full, saving to cloud primary."); }

        if (supabaseClient) {
            try {
                const timeout = new Promise((_, reject) => setTimeout(() => reject(new Error('timeout')), 6000));
                await Promise.race([ supabaseClient.from('audits').insert([newAudit]), timeout ]);
            } catch(e) {
                console.warn("Supabase insert warning:", e.message);
            }
        }

        alert(`✅ Formal TW Report Submitted!\nCompliance Score: ${score}%\n${newAudit.pdfUrl ? 'PDF uploaded to Supabase cloud.' : ''}`);
        window.location.href = 'officerdashboard.html';

    } catch (err) {
        alert(`❌ Submission Error: ${err.message}`);
    } finally {
        btn.innerHTML = originalText;
        btn.disabled = false;
        lucide.createIcons();
    }
}
